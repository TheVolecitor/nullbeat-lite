NullBeat API

Cache Integration

How to verify and stream cached tracks from TorBox.

01. Auth

Tidal Auth Request

Fetch a Client Credentials token from Tidal. Requires a Tidal Developer Client ID and Secret.

POST https://auth.tidal.com/v1/oauth2/token
Headers:
  Authorization: Basic <base64(CLIENT_ID:CLIENT_SECRET)>
  Content-Type: application/x-www-form-urlencoded

Body:
  grant_type=client_credentialsHTTP
02. Search

Tidal Search Request

Search Tidal's OpenAPI v2. You must include specific relationship metadata to get artists and album art.

GET https://openapi.tidal.com/v2/searchResults
Query Parameters:
  filter[query]: "Monkeys Spinning Monkeys Kevin MacLeod"
  countryCode: "US"
  include: "tracks.artists,tracks.albums.coverArt,albums.artists,albums.coverArt,artists.profileArt"

Headers:
  Authorization: Bearer <ACCESS_TOKEN>
  Accept: application/vnd.api+jsonHTTP

Tidal returns the relationships.tracks.data array natively ordered by relevance. Pick the top 2 tracks and map them to their full objects using the included array.

03. Format

Deterministic Filenames

TorBox caches are verified using MD5 hashes of standard filenames.

{Clean Title} - {Clean Primary Artist} [NullBeat {Tidal Track ID} - {Quality Label} ].ext

Example for Monkeys Spinning Monkeys by Kevin MacLeod (Tidal Track ID: 359051850):

Monkeys Spinning Monkeys - Kevin MacLeod [NullBeat 359051850 - Hi-Res].flac
Monkeys Spinning Monkeys - Kevin MacLeod [NullBeat 359051850 - HiFi CD].flac
Monkeys Spinning Monkeys - Kevin MacLeod [NullBeat 359051850 - 320kbps].m4a
Monkeys Spinning Monkeys - Kevin MacLeod [NullBeat 359051850 - 96kbps].m4a

Note: Sanitize the title and artist by stripping characters: / \ ? % * : | " < >

04. Cache

TorBox Cache Check

Hash the generated filenames using MD5 and POST them to TorBox.

POST https://api.torbox.app/v1/api/torrents/checkcached
Query Parameters:
  format: "object"
  list_files: "false"

Headers:
  Authorization: Bearer <TORBOX_API_KEY>
  Content-Type: application/json

Body:
  {
    "hashes": ["md5hex1", "md5hex2", ...]
  }HTTP
05. Demos

Code Examples

Reference implementations. Create a .env file in your project directory with your credentials: 

TIDAL_CLIENT_ID=your_tidal_client_id_here
TIDAL_CLIENT_SECRET=your_tidal_client_secret_here
TORBOX_API_KEY=your_torbox_api_key_here.env
import crypto from 'node:crypto';

const CLIENT_ID = process.env.TIDAL_CLIENT_ID;
const CLIENT_SECRET = process.env.TIDAL_CLIENT_SECRET;
const TORBOX_API_KEY = process.env.TORBOX_API_KEY;

if (!CLIENT_ID || !CLIENT_SECRET || !TORBOX_API_KEY) {
  console.error("Missing credentials in environment variables.");
  process.exit(1);
}

// 1. Auth
const authData = Buffer.from(`${CLIENT_ID}:${CLIENT_SECRET}`).toString('base64');
const tokenRes = await fetch('https://auth.tidal.com/v1/oauth2/token', {
  method: 'POST',
  headers: { Authorization: `Basic ${authData}`, 'Content-Type': 'application/x-www-form-urlencoded' },
  body: 'grant_type=client_credentials',
});
const { access_token } = await tokenRes.json();

// 2. Search
const searchRes = await fetch(
  `https://openapi.tidal.com/v2/searchResults?filter[query]=${encodeURIComponent('Monkeys Spinning Monkeys Kevin MacLeod')}&countryCode=US&include=tracks.artists,tracks.albums.coverArt`,
  { headers: { Authorization: `Bearer ${access_token}`, Accept: 'application/vnd.api+json' } }
);
const json = await searchRes.json();

// 3. Parse map based on relationships
const includedMap = new Map(json.included.map(i => [`${i.type}:${i.id}`, i]));
const trackRefs = json.data?.[0]?.relationships?.tracks?.data || [];
const tracks = trackRefs
  .slice(0, 2)
  .map(ref => includedMap.get(`tracks:${ref.id}`))
  .filter(Boolean);

// 4. Build hashes & print candidates
const clean = s => (s || '').replace(/[\/\\?%*:|"<>]/g, '').trim();
const tiers = [{ label: 'Hi-Res', ext: 'flac' }, { label: 'HiFi CD', ext: 'flac' }, { label: '320kbps', ext: 'm4a' }, { label: '96kbps', ext: 'm4a' }];
const filenamesSet = new Set();
const hashes = [];

console.log('Checking candidate filenames & MD5 hashes:');
for (const t of tracks) {
  const artistNames = (t.relationships?.artists?.data || [])
    .map(a => includedMap.get(`artists:${a.id}`)?.attributes?.name).filter(Boolean);
  const id = String(t.id).replace(/^track:/, '');
  for (const tier of tiers) {
    const fn = `${clean(t.attributes?.title)} - ${clean(artistNames[0])} [NullBeat ${id} - ${tier.label}].${tier.ext}`;
    const md5 = crypto.createHash('md5').update(fn).digest('hex');
    filenamesSet.add(fn);
    hashes.push(md5);
    console.log(`  ${md5} -> ${fn}`);
  }
}

// 5. Check TorBox
const tbRes = await fetch('https://api.torbox.app/v1/api/torrents/checkcached?format=object&list_files=false', {
  method: 'POST',
  headers: { Authorization: `Bearer ${TORBOX_API_KEY}`, 'Content-Type': 'application/json' },
  body: JSON.stringify({ hashes }),
});
const { data } = await tbRes.json();

// 6. Result (TorBox keys response by infohash, match by name)
const hits = Object.values(data || {}).map(i => i?.name).filter(n => filenamesSet.has(n));
console.log(hits.length ? `✓ CACHED: ${hits[0]}` : '✗ Not cached yet');
import os, requests, hashlib, base64

# Load .env file automatically
try:
    from dotenv import load_dotenv
    load_dotenv()
except ImportError:
    if os.path.exists('.env'):
        with open('.env', 'r') as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith('#') and '=' in line:
                    k, v = line.split('=', 1)
                    os.environ.setdefault(k.strip(), v.strip().strip('"\''))

CLIENT_ID = os.environ.get('TIDAL_CLIENT_ID')
CLIENT_SECRET = os.environ.get('TIDAL_CLIENT_SECRET')
TORBOX_KEY = os.environ.get('TORBOX_API_KEY')

if not CLIENT_ID or not CLIENT_SECRET or not TORBOX_KEY:
    print("Missing credentials in environment variables.")
    exit(1)

# 1. Auth
creds = base64.b64encode(f"{CLIENT_ID}:{CLIENT_SECRET}".encode()).decode()
auth_res = requests.post('https://auth.tidal.com/v1/oauth2/token', 
    headers={'Authorization': f'Basic {creds}'}, 
    data={'grant_type': 'client_credentials'})
token = auth_res.json()['access_token']

# 2. Search
search_res = requests.get('https://openapi.tidal.com/v2/searchResults',
    headers={'Authorization': f'Bearer {token}', 'Accept': 'application/vnd.api+json'},
    params={'filter[query]': 'Monkeys Spinning Monkeys Kevin MacLeod', 'countryCode': 'US', 'include': 'tracks.artists'})
data = search_res.json()

# 3. Parse included map based on relationships
included = { f"{i['type']}:{i['id']}": i for i in data.get('included', []) }
track_refs = data['data'][0].get('relationships', {}).get('tracks', {}).get('data', [])[:2]

tracks = []
for t_ref in track_refs:
    t = included.get(f"tracks:{t_ref['id']}")
    if t: tracks.append(t)

# 4. Build hashes & print candidates
def clean(s): return "".join([c for c in (s or "") if c not in r'/\?%*:|"<>']).strip()
tiers = [{'label': 'Hi-Res', 'ext': 'flac'}, {'label': 'HiFi CD', 'ext': 'flac'}, {'label': '320kbps', 'ext': 'm4a'}, {'label': '96kbps', 'ext': 'm4a'}]
hashes = []
filenames_set = set()

print("Checking candidate filenames & MD5 hashes:")
for t in tracks:
    art_refs = t.get('relationships', {}).get('artists', {}).get('data', [])
    artist = included.get(f"artists:{art_refs[0]['id']}", {}).get('attributes', {}).get('name', '') if art_refs else ''
    tid = str(t['id']).replace('track:', '')
    
    for tier in tiers:
        fn = f"{clean(t['attributes']['title'])} - {clean(artist)} [NullBeat {tid} - {tier['label']}].{tier['ext']}"
        md5 = hashlib.md5(fn.encode()).hexdigest()
        filenames_set.add(fn)
        hashes.append(md5)
        print(f"  {md5} -> {fn}")

# 5. Check TorBox
tb_res = requests.post('https://api.torbox.app/v1/api/torrents/checkcached',
    headers={'Authorization': f'Bearer {TORBOX_KEY}'},
    params={'format': 'object', 'list_files': 'false'},
    json={'hashes': hashes})

# 6. Result (TorBox keys response by infohash, match by name)
tb_data = tb_res.json().get('data', {})
hits = [item.get('name') for item in tb_data.values() if isinstance(item, dict) and item.get('name') in filenames_set]
print("\nCached Hits:", hits)
